+// FDE-006R: stable, human-readable identity for marker objects. LOG ONLY.
+inline std::string fde006rObjLabel(const frBlockObject* obj)
+{
+  if (obj == nullptr) {
+    return "null";
+  }
+  std::string name;
+  switch (obj->typeId()) {
+    case frcNet:
+      name = static_cast<const frNet*>(obj)->getName();
+      break;
+    case frcInst:
+      name = static_cast<const frInst*>(obj)->getName();
+      break;
+    case frcInstTerm:
+      name = static_cast<const frInstTerm*>(obj)->getName();
+      break;
+    default:
+      break;
+  }
+  return name + "#" + std::to_string(static_cast<int>(obj->typeId())) + ":"
+         + std::to_string(obj->getId());
+}
+inline std::string fde006rSrcsLabel(const frOrderedIdSet<frBlockObject*>& srcs)
+{
+  std::string out;
+  for (const auto* obj : srcs) {
+    if (!out.empty()) {
+      out += ",";
+    }
+    out += fde006rObjLabel(obj);
+  }
+  return out;
+}
+inline std::string fde006rPairsLabel(
+    const std::vector<std::pair<frBlockObject*,
+                                std::tuple<frLayerNum, odb::Rect, bool>>>& objs)
+{
+  std::string out;
+  for (const auto& pair : objs) {
+    if (!out.empty()) {
+      out += ",";
+    }
+    out += fde006rObjLabel(pair.first);
+    out += "@L" + std::to_string(static_cast<int>(std::get<0>(pair.second)));
+  }
+  return out;
+}
